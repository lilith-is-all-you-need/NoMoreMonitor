#include "messager.h"
#include "notice.h"

HANDLE event;
HANDLE mapping;
HANDLE mutex;
HANDLE quit_event;
void* ptr_buf;
HANDLE msg_thread;

DWORD WINAPI lililth_thread_proc(LPVOID lpParameter) {
    HANDLE waits[2] = { quit_event, event };  // 注意顺序：quit 优先
    for (;;) {
        DWORD r = WaitForMultipleObjects(2, waits, FALSE, INFINITE);
        if (r == WAIT_OBJECT_0) {
            // 收到退出信号
            break;
        }
        if (r != WAIT_OBJECT_0 + 1) {
            // 异常，直接退出免得死循环
            break;
        }

        // —— 受 mutex 保护读取共享内存 ——
        struct log_data curr_pack = { 0 };
        if (WaitForSingleObject(mutex, 2000) == WAIT_OBJECT_0) {
            memcpy(&curr_pack, ptr_buf, sizeof(struct log_data));
            ReleaseMutex(mutex);
        }
        else {
            continue; // 拿不到锁就放弃这次（避免被卡死）
        }

        switch (curr_pack.curr_state) {
        case status_error:
            wprintf(L"[!]Log from Dll:%s Err code:%d\n",
                curr_pack.log_buffer, curr_pack.pot_error);
            break;
        case status_waiting:
            break;
        case status_watching:
            if (hwnd) PostMessage(hwnd, WM_IPC_STATE, status_watching, 0);
            break;
        case status_start:
            wprintf(L"[-]Spy start!\n");
            if (hwnd) PostMessage(hwnd, WM_IPC_STATE, status_start, 0);
            break;
        case status_stop:
            wprintf(L"[-]Spy stop!\n");
            if (hwnd) PostMessage(hwnd, WM_IPC_STATE, status_stop, 0);
            break;
        case status_log:
            wprintf(L"[*]Log:%s\n", curr_pack.log_buffer);
            // 新增：把倒计时文本传递给 UI 窗口，让它刷新屏幕字样
            // 注意：这里必须使用 SendMessage (同步阻塞)，以保证 curr_pack 内存安全拷贝完毕后再继续
            if (hwnd) {
                SendMessage(hwnd, WM_IPC_STATE, status_log, (LPARAM)curr_pack.log_buffer);
            }
            break;
        case status_info:
            wprintf(L"[*]Info:%s\n", curr_pack.log_buffer);
            break;
        default:
            break;
        }
    }
    return 0;
}


bool messager_init(void) {
    SECURITY_DESCRIPTOR sd;
    InitializeSecurityDescriptor(&sd, SECURITY_DESCRIPTOR_REVISION);
    SetSecurityDescriptorDacl(&sd, TRUE, NULL, FALSE); // 允许所有人访问

    SECURITY_ATTRIBUTES sa = { 0 };
    sa.nLength = sizeof(sa);
    sa.lpSecurityDescriptor = &sd;
    sa.bInheritHandle = FALSE;
    // 1) Event
    event = CreateEventW(NULL, FALSE, FALSE, L"Local\\LilithLogEvent");
    if (!event) { wprintf(L"[!]can't create event (%lu)\n", GetLastError()); goto fail; }

    // 2) Mutex
    mutex = CreateMutexW(NULL, FALSE, L"Local\\LilithMutex");
    if (!mutex) { wprintf(L"[!]can't create mutex (%lu)\n", GetLastError()); goto fail; }

    // 3) FileMapping
    mapping = CreateFileMappingW(INVALID_HANDLE_VALUE, NULL, PAGE_READWRITE,
        0, sizeof(struct log_data), L"Local\\LilithSharedMem");
    if (!mapping) { wprintf(L"[!]can't create file mapping (%lu)\n", GetLastError()); goto fail; }

    // 4) MapView
    ptr_buf = MapViewOfFile(mapping, FILE_MAP_ALL_ACCESS, 0, 0, sizeof(struct log_data));
    if (!ptr_buf) { wprintf(L"[!]can't map view (%lu)\n", GetLastError()); goto fail; }
    ZeroMemory(ptr_buf, sizeof(struct log_data));

    // 5) Quit event (内部使用，不命名)
    quit_event = CreateEventW(NULL, TRUE, FALSE, NULL);
    if (!quit_event) { wprintf(L"[!]can't create quit_event (%lu)\n", GetLastError()); goto fail; }

    // 6) Worker thread
    msg_thread = CreateThread(NULL, 0, lililth_thread_proc, NULL, 0, NULL);
    if (!msg_thread) { wprintf(L"[!]can't create messager thread (%lu)\n", GetLastError()); goto fail; }

    return true;

fail:
    messager_uninit();   // 统一清理，幂等
    return false;
}


void messager_uninit(void) {
    // 通知线程退出
    if (quit_event) SetEvent(quit_event);

    // 等线程结束（最多等 3 秒，避免被卡）
    if (msg_thread) {
        if (WaitForSingleObject(msg_thread, 3000) == WAIT_TIMEOUT) {
            TerminateThread(msg_thread, 0); // 万不得已
        }
        CloseHandle(msg_thread);
        msg_thread = NULL;
    }

    if (ptr_buf) { UnmapViewOfFile(ptr_buf); ptr_buf = NULL; }
    if (mapping) { CloseHandle(mapping);     mapping = NULL; }
    if (mutex) { CloseHandle(mutex);       mutex = NULL; }
    if (event) { CloseHandle(event);       event = NULL; }
    if (quit_event) { CloseHandle(quit_event);  quit_event = NULL; }
}
